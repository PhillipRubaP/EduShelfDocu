\chapter*{Abkürzungsverzeichnis}
\addcontentsline{toc}{chapter}{Abkürzungsverzeichnis}

\begin{longtable}{@{}p{3.5cm}p{12cm}@{}}

\textbf{ANN}    & Approximate Nearest Neighbor -- Algorithmus zur näherungsweisen Suche des nächsten Nachbarn in hochdimensionalen Vektorräumen \\[4pt]
\textbf{API}    & Application Programming Interface -- Programmierschnittstelle zur Kommunikation zwischen Softwarekomponenten \\[4pt]
\textbf{ASP}    & Active Server Pages -- Webframework von Microsoft (hier: ASP.NET Core) \\[4pt]
\textbf{BLOB}   & Binary Large Object -- Binäres Datenobjekt zur Speicherung unstrukturierter Daten \\[4pt]
\textbf{CORS}   & Cross-Origin Resource Sharing -- Mechanismus zur kontrollierten Freigabe von Ressourcen über Domaingrenzen hinweg \\[4pt]
\textbf{CRUD}   & Create, Read, Update, Delete -- Die vier grundlegenden Datenbankoperationen \\[4pt]
\textbf{CSS}    & Cascading Style Sheets -- Stylesheet-Sprache zur Gestaltung von Webseiten \\[4pt]
\textbf{CUDA}   & Compute Unified Device Architecture -- Nvidia-Plattform für parallele Berechnungen auf GPUs \\[4pt]
\textbf{DI}     & Dependency Injection -- Entwurfsmuster zur Entkopplung von Abhängigkeiten \\[4pt]
\textbf{DOCX}   & Document XML -- Dateiformat für Microsoft Word-Dokumente \\[4pt]
\textbf{DOM}    & Document Object Model -- Programmierschnittstelle zur Darstellung und Manipulation von HTML-Dokumenten \\[4pt]
\textbf{DTO}    & Data Transfer Object -- Datenobjekt zum Transport von Daten zwischen Softwareschichten \\[4pt]
\textbf{EF}     & Entity Framework -- Object-Relational Mapper für .NET (hier: EF Core) \\[4pt]
\textbf{ERD}    & Entity Relationship Diagram -- Diagramm zur Darstellung von Datenbank-Entitäten und deren Beziehungen \\[4pt]
\textbf{GPU}    & Graphics Processing Unit -- Grafikprozessor, hier genutzt für KI-Berechnungen \\[4pt]
\textbf{HNSW}   & Hierarchical Navigable Small World -- Indexstruktur für effiziente Vektorsuche \\[4pt]
\textbf{HTML}   & Hypertext Markup Language -- Auszeichnungssprache für Webseiten \\[4pt]
\textbf{HTTP}   & Hypertext Transfer Protocol -- Übertragungsprotokoll für das Web \\[4pt]
\textbf{HTTPS}  & Hypertext Transfer Protocol Secure -- Verschlüsselte Variante von HTTP \\[4pt]
\textbf{JSON}   & JavaScript Object Notation -- Leichtgewichtiges Datenaustauschformat \\[4pt]
\textbf{JWT}    & JSON Web Token -- Standard für tokenbasierte Authentifizierung \\[4pt]
\textbf{KI}     & Künstliche Intelligenz \\[4pt]
\textbf{LLM}    & Large Language Model -- Großes Sprachmodell, z.\,B. Llama 3 oder Qwen3 \\[4pt]
\textbf{OCR}    & Optical Character Recognition -- Optische Zeichenerkennung zur Textextraktion aus Bildern \\[4pt]
\textbf{ORM}    & Object-Relational Mapper -- Werkzeug zur Abbildung von Datenbanktabellen auf Objekte \\[4pt]
\textbf{PDF}    & Portable Document Format -- Plattformunabhängiges Dokumentenformat \\[4pt]
\textbf{RAG}    & Retrieval-Augmented Generation -- Verfahren zur Anreicherung von LLM-Antworten durch externe Wissensquellen \\[4pt]
\textbf{RBAC}   & Role-Based Access Control -- Rollenbasierte Zugriffskontrolle \\[4pt]
\textbf{REST}   & Representational State Transfer -- Architekturstil für Web-APIs \\[4pt]
\textbf{S3}     & Simple Storage Service -- Objektspeicher-Standard (Amazon, hier: MinIO-kompatibel) \\[4pt]
\textbf{SDK}    & Software Development Kit -- Sammlung von Entwicklungswerkzeugen \\[4pt]
\textbf{SMTP}   & Simple Mail Transfer Protocol -- Protokoll zum Versenden von E-Mails \\[4pt]
\textbf{SPA}    & Single Page Application -- Webanwendung, die nur aus einer einzigen HTML-Seite besteht \\[4pt]
\textbf{SQL}    & Structured Query Language -- Abfragesprache für relationale Datenbanken \\[4pt]
\textbf{TXT}    & Textdatei -- Einfaches, unformatiertes Dateiformat \\[4pt]
\textbf{UI}     & User Interface -- Benutzeroberfläche \\[4pt]
\textbf{URL}    & Uniform Resource Locator -- Eindeutige Adresse einer Ressource im Web \\[4pt]
\textbf{UX}     & User Experience -- Benutzererfahrung \\[4pt]
\textbf{VRAM}   & Video Random Access Memory -- Grafikspeicher der GPU \\[4pt]
\textbf{XML}    & Extensible Markup Language -- Erweiterbare Auszeichnungssprache \\

\end{longtable}
